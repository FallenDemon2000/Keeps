07 — Implementation Notes

Font Import

Place these at the very top of src/index.css before all other statements.

@import url('https://fonts.googleapis.com/css2?family=DM+Sans:wght@300;400;500;600;700&display=swap');
@import url('https://fonts.googleapis.com/css2?family=JetBrains+Mono:wght@400;500;600&display=swap');
@import 'tailwindcss';

/* Then declare font variables */
@theme inline {
  --font-sans: 'DM Sans', sans-serif;
  --font-mono: 'JetBrains Mono', monospace;
}

Screen Architecture

// State machine: 3 screens, bottom tab navigation
type Screen = "upload" | "results" | "settings"

// Upload → (scan) → Results
// Results badge = groups.length
// Settings reachable from any screen

// Duplicate detection: perceptual hashing
// 1. Load each File as ObjectURL
// 2. Draw onto 8×8 canvas (ctx.drawImage)
// 3. Compute grayscale → average hash (64 bits)
// 4. Hamming distance between all pairs
// 5. Group pairs where similarity ≥ threshold

similarity = (1 - hammingDistance / 64) * 100
